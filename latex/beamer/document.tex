% 2 Contexto
\begin{frame}{Contexto}
  \begin{itemize}
    \item Volume crescente em \textit{data lakes} e \textit{lakehouses} exige melhor aproveitamento no consumo dos dados por consulta
    \item \textit{Apache Iceberg} como formato de tabela com metadados hierárquicos
    \item \textit{Trino} como \textit{engine} de consultas distribuído
    \item Em geral, o \textit{cluster} de execução é escolhido de forma estática
  \end{itemize}
  \begin{alertblock}{Problema}
    Consultas leves ocupam \textit{clusters} robustos e consultas pesadas falham ou degradam em \textit{clusters} pequenos, com aproveitamento ineficiente dos recursos de processamento.
  \end{alertblock}
\end{frame}

% 3 Motivacao
\begin{frame}{Motivação}
  \begin{itemize}
    \item Provisionamento excessivo eleva custos operacionais e financeiros, além de desperdiçar recursos computacionais
    \item Subdimensionamento aumenta falhas e tempo de resposta
    \item Metadados do \textit{Iceberg} permitem estimar o custo \textbf{antes} da execução
    \item Lacuna: roteamento entre \textit{clusters} \textit{Trino} de capacidades distintas
  \end{itemize}
\end{frame}

% 4 Objetivos
\begin{frame}{Objetivos}
  \begin{block}{Objetivo Geral}
    Desenvolver o \textit{DyraSQL}, mecanismo que usa \textit{EXPLAIN (TYPE IO)} para definir automaticamente o \textit{cluster} mais adequado para cada consulta SQL.
  \end{block}
  \begin{enumerate}
    \item Mapear metadados \textit{Iceberg} úteis para estimar custo
    \item Definir regras de decisão
    \item Construir protótipo local
    \item Realizar experimentos controlados
  \end{enumerate}
\end{frame}

% 5 Referencial teorico
\begin{frame}{Referencial Teórico}
  \begin{itemize}
    \item \textit{Apache Iceberg}: \textit{snapshots}, manifestos e estatísticas por arquivo (\textit{data skipping})
    \item \textit{Trino}: \textit{EXPLAIN (TYPE IO)} retorna \textit{outputSizeInBytes} e linhas estimadas
    \item \textit{Trino Gateway}: balanceamento e roteamento entre \textit{clusters}
    \item \textit{Lakehouse}: unificação de \textit{data lake} e \textit{warehouse}
  \end{itemize}
  \begin{alertblock}{Trabalhos relacionados}
    Strausz et al. (2025) selecionam o \textit{engine} SQL. O \textit{DyraSQL} mantém o mesmo \textit{engine} (\textit{Trino}) e seleciona o \textit{cluster} com o dimensionamento mais adequado (\textit{small}, \textit{medium} ou \textit{large}).
  \end{alertblock}
\end{frame}

% 7 Algoritmo
\begin{frame}{Metodologia - Algoritmo de Decisão}
  \begin{block}{Score}
    \begin{equation} S = w_1\, f_v + w_2\, f_c + w_3\, f_h \end{equation}
    {\small
    \begin{itemize}\setlength{\itemsep}{0.15em}
      \item $f_v$: volume efetivo da consulta
      \item $f_c$: complexidade estrutural
      \item $f_h$: histórico de execuções
      \item $w_1{=}0{,}5$, $w_2{=}0{,}3$, $w_3{=}0{,}2$ nos testes, ajustáveis ao ambiente
    \end{itemize}}
  \end{block}
  \begin{alertblock}{Roteamento}
    $S<0{,}3$: \textit{small} \quad $0{,}3\le S\le 0{,}7$: \textit{medium} \quad $S>0{,}7$: \textit{large}\\[0.35em]
    {\small Decisões em \textit{cache} por \textit{fingerprint}, com \textit{TTL} de 24 horas.}
  \end{alertblock}
\end{frame}

% 8 Exemplo de calculo
\begin{frame}{Metodologia - Exemplo de Cálculo}
  \begin{block}{Fator volume}
    \begin{equation*}
      f_v = \left( \frac{\log(T_e)}{\log(T_{max})} \times 0{,}6 + \frac{\log(R_e)}{\log(R_{max})} \times 0{,}4 \right) \times (1 - F_o)
    \end{equation*}
  \end{block}
  {\small
  \begin{itemize}
    \item $T_e = 1{,}25$~GB e $R_e = 2{,}5 \times 10^{6}$ vêm do \textit{EXPLAIN (TYPE IO)} da consulta
    \item $T_{max}$ e $R_{max}$ são o volume máximo do \textit{warehouse} de cada ambiente (no exemplo, 1~TB e $10^{9}$ linhas)
  \end{itemize}
  \begin{equation*}
    f_v = \Bigl(\tfrac{\log(1{,}25)}{\log(1000)}\times 0{,}6 + \tfrac{\log(2500000)}{\log(10^{9})}\times 0{,}4\Bigr)\times 0{,}9 = 0{,}27
  \end{equation*}
  }
\end{frame}

\begin{frame}{Metodologia - Exemplo de Cálculo}
  \begin{block}{Fator complexidade}
    \begin{equation*}
      f_c = \frac{1\times 0{,}2 + 3\times 0{,}15 + 2\times 0{,}25 + 1\times 0{,}02 + 2\times 0{,}1}{2{,}0} = 0{,}69
    \end{equation*}
    {\small
    \begin{itemize}\setlength{\itemsep}{0.1em}
      \item 1 \textit{JOIN} ($0{,}2$), 3 agregações ($0{,}15$), 2 subconsultas ($0{,}25$)
      \item 1 filtro particionado ($0{,}02$) e 2 não particionados ($0{,}1$), soma ponderada $1{,}37$
      \item $L_c{=}2$ porque a soma $1{,}37$ não pode saturar em $1$, nem perder peso com teto grande
    \end{itemize}}
  \end{block}
  \begin{block}{Fator histórico}
    {\small
    \begin{itemize}\setlength{\itemsep}{0.1em}
      \item sem histórico, $f_h = 0{,}5$ (caso do exemplo, neutro)
      \item sucesso, $f_h = S_p$ (repete a faixa que funcionou)
      \item falha em \textit{small}/\textit{medium}, $f_h = 1 - S_p$ (busca mais capacidade)
      \item falha no \textit{large}, $f_h = 0{,}5$ (não reduz, não é subdimensionamento)
    \end{itemize}}
  \end{block}
\end{frame}

% 13 Resumo dos resultados
\begin{frame}{Resultados - Síntese}
  \begin{block}{Consulta de referência}
    \begin{equation*}
      S = 0{,}5 \times 0{,}27 + 0{,}3 \times 0{,}69 + 0{,}2 \times 0{,}5 = 0{,}44
    \end{equation*}
    Volume $f_v{=}0{,}27$, complexidade $f_c{=}0{,}69$ e histórico $f_h{=}0{,}5$. Com $0{,}3 \le S \le 0{,}7$, a consulta segue para o \textit{cluster medium}.
  \end{block}
  \begin{itemize}
    \item Consultas simples e com filtros estreitos tendem ao \textit{cluster small}
    \item \textit{EXPLAIN (TYPE IO)} fornece volume compatível com os filtros de partição
    \item Reexecuções recuperam a decisão do \textit{cache}
  \end{itemize}
  \vspace{0.3em}
  {\footnotesize Limitação: \textit{dataset} sintético, sem elasticidade de nuvem.}
